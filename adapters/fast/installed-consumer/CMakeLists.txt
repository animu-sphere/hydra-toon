cmake_minimum_required(VERSION 3.24)
project(ToonAvatarConsumer LANGUAGES CXX)
if(AVATAR_COMPONENT)
  find_package(Toon REQUIRED COMPONENTS AvatarState)
  add_executable(consumer main.cpp)
  target_link_libraries(consumer PRIVATE Toon::AvatarState)
else()
  find_package(Toon REQUIRED)
  if(TARGET Toon::AvatarState OR TARGET Toon::AvatarRuntimeHeaders)
    message(FATAL_ERROR "Default Toon package loaded optional runtime dependencies")
  endif()
  add_executable(consumer main.cpp)
  target_link_libraries(consumer PRIVATE Toon::RenderWorld)
endif()
target_compile_features(consumer PRIVATE cxx_std_20)
target_compile_definitions(consumer PRIVATE AVATAR_COMPONENT=$<BOOL:${AVATAR_COMPONENT}>)
